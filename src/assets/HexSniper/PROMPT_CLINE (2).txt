import { HexSniperController } from './HexSniperController.js';
import { HexSniperAttacks } from './HexSniperAttacks.js';

/** Local authoritative game. Multiplayer: run gameplay on the server and
 * drive client visuals from validated events and world-space tip positions. */
export function createHexSniper({ gltf, holder, effectsParent, world, pose,
  ownerId, onEvent, tuning = {} }) {
  const visuals = new HexSniperController(gltf, { effectsParent });
  holder.add(visuals.object);
  visuals.object.updateWorldMatrix(true, true);
  let attacks;
  try {
    attacks = new HexSniperAttacks({ ...tuning, world, pose, ownerId, onEvent, visuals });
  } catch (error) {
    visuals.dispose();
    throw error;
  }
  return {
    object: visuals.object, visuals, attacks,
    primary: () => attacks.tryTongue(),
    secondary: () => attacks.tryBite(),
    fixedUpdate: dt => attacks.update(dt),
    updateVisuals: dt => visuals.update(dt),
    cancel: reason => attacks.cancel(reason),
    dispose() { attacks.dispose(); visuals.dispose(); }
  };
}

/** Optional mouse/touch bindings. Requests may also be network commands.
 * enabled() follows the game's equipped/paused/UI state. Returns cleanup. */
export function bindHexSniperInputs({ element, requestTongue, requestBite,
  tongueButton = null, biteButton = null, enabled = () => true }) {
  const tongue = () => { if (enabled()) requestTongue(); };
  const bite = () => { if (enabled()) requestBite(); };
  const pointer = event => {
    if (!enabled() || event.pointerType === 'touch') return;
    if (event.button === 0) { event.preventDefault(); requestTongue(); }
    if (event.button === 2) { event.preventDefault(); requestBite(); }
  };
  const context = event => { if (enabled()) event.preventDefault(); };
  element.addEventListener('pointerdown', pointer);
  element.addEventListener('contextmenu', context);
  tongueButton?.addEventListener('click', tongue);
  biteButton?.addEventListener('click', bite);
  return () => {
    element.removeEventListener('pointerdown', pointer);
    element.removeEventListener('contextmenu', context);
    tongueButton?.removeEventListener('click', tongue);
    biteButton?.removeEventListener('click', bite);
  };
}
